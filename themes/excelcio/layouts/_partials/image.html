{{- /* Responsive image: AVIF + WebP + original, srcset, explicit dimensions (SVG and GIF served unchanged). Context: dict "page" page (to resolve page resources) "src" path of a page resource or a global asset "alt" alternative text (required unless "decorative" is true) "decorative" bool "sizes" sizes attribute (default "100vw") "widths" slice of widths (default 480 800 1200 1600) "priority" bool: LCP image (eager, fetchpriority high) "class" */ -}} {{- $img := "" -}} {{- with .page }}{{ $img = .Resources.Get $.src }}{{ end -}} {{- if not $img }}{{ $img = resources.Get .src }}{{ end -}} {{- if and (not .alt) (not .decorative) }} {{- errorf "image.html : texte alternatif manquant pour %q" .src }} {{- end -}} {{- with $img }} {{- if eq .MediaType.SubType "svg" }} {{- else if eq .MediaType.SubType "gif" }} {{- /* GIFs are served as is: resizing decodes every frame of an animation (memory exhaustion on long screen captures) and AVIF/WebP drop it. */}} {{- else }} {{- $widths := $.widths | default (slice 480 800 1200 1600) }} {{- $sizes := $.sizes | default "100vw" }} {{- $fallback := .Resize (printf "%dx webp" (math.Min 1200 .Width | int)) }} {{- range slice "avif" "webp" }} {{- $fmt := . }} {{- /* Requested widths up to the original one, never upscaled; the original width (capped at the largest requested one) is always offered. */}} {{- $top := math.Min $img.Width (index (sort $widths) (sub (len $widths) 1)) | int }} {{- $candidates := slice }} {{- range $widths }}{{ if lt . $top }}{{ $candidates = $candidates | append . }}{{ end }}{{ end }} {{- $candidates = $candidates | append $top }} {{- $set := slice }} {{- range $candidates }} {{- $r := $img.Resize (printf "%dx %s" . $fmt) }} {{- $set = $set | append (printf "%s %dw" $r.RelPermalink $r.Width) }} {{- end }} {{- end }} {{- end }} {{- else }} {{- warnf "image.html : image introuvable %q" .src }} {{- end -}}
