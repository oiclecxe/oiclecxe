{{- /* Social image (Open Graph, 1200 × 630) of a page. - The page image (`image` front matter, page resource) when there is one (not GIF or SVG); - otherwise an image generated by Hugo: brand background (site.Params.style.brand), content type and page title in Geist (TrueType copy used at build time only), site name at the bottom. Placeholder titles fall back to the site title. Context: page. Returns: image resource, or nil. */ -}} {{- $img := "" -}} {{- with .Params.image }}{{ with $.Resources.Get . }}{{ if not (in (slice "gif" "svg") .MediaType.SubType) }}{{ $img = .Fill "1200x630 Center jpg q82" }}{{ end }}{{ end }}{{ end -}} {{- if not $img }} {{- $style := site.Params.style }} {{- $font := resources.Get "fonts/Geist-Medium.ttf" }} {{- with resources.Get "images/og-base.png" }} {{- $eyebrow := "" }} {{- if $.IsPage }}{{ with partial "func/type-label.html" $.Type }}{{ $eyebrow = upper . }}{{ end }}{{ end }} {{- $title := partial "func/real.html" $.Title | default site.Title }} {{- if $.IsHome }}{{ $title = partial "func/real.html" (site.Params.footer.headline | default "") | default site.Title }}{{ end }} {{- $bg := $style.brand }} {{- $fg := index $style "light-1" }} {{- /* Canvas 1112 × 630 (text wraps at its width), right margin of 88 px added last. */}} {{- $filters := slice (images.Padding 0 1112 630 0 $bg) (images.Padding -1 0 0 -1 $bg) }} {{- with $eyebrow }} {{- $filters = $filters | append (images.Text . (dict "font" $font "size" 26 "color" $fg "x" 88 "y" 96)) }} {{- end }} {{- $filters = $filters | append (images.Text $title (dict "font" $font "size" 76 "linespacing" 10 "color" $fg "x" 88 "y" 150)) }} {{- if ne $title site.Title }} {{- $filters = $filters | append (images.Text site.Title (dict "font" $font "size" 30 "color" $fg "x" 88 "y" 534)) }} {{- end }} {{- $filters = $filters | append (images.Padding 0 88 0 0 $bg) }} {{- $img = .Filter $filters }} {{- end }} {{- end -}} {{- return $img -}}
